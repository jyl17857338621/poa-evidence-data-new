function normTxt(t){return (t||'').toString().toLowerCase().trim();}
function slugTxt(t){return normTxt(t).replace(/[_\-\s\.]+/g,'');}
function tokenHead(t){const n=normTxt(t);const m=n.match(/^([a-z0-9]+)/);return m?m[1]:'';}
function scoreSpuMatch(text,spu){
  const n=normTxt(text), s=slugTxt(text), head=tokenHead(text);
  if(!n)return 0;
  const code=normTxt(spu.code), name=normTxt(spu.name), codeSlug=slugTxt(spu.code), nameSlug=slugTxt(spu.name);
  // exact
  if(code===n||name===n||codeSlug===s||nameSlug===s)return 1000+code.length;
  // leading token (e.g. "SPU_瀹夎璇存槑.pdf" / "SPU 瀹夎璇存槑.pdf")
  if(head && (head===code||head===name||head===codeSlug||head===nameSlug))return 900+code.length;
  // contains
  let score=0;
  const candidates=[code,name,codeSlug,nameSlug].filter(x=>x.length>=3);
  for(const c of candidates){
    if(n.includes(c)||s.includes(c))score=Math.max(score,500+c.length);
  }
  return score;
}
function findSpuByText(text){
  if(!spus||!spus.length)return null;
  const n=normTxt(text);
  if(!n)return null;
  let best=null,bestScore=0;
  for(const spu of spus){
    const sc=scoreSpuMatch(text,spu);
    if(sc>bestScore){bestScore=sc;best=spu;}
    else if(sc&&sc===bestScore && spu.code.length>(best?best.code.length:0)){best=spu;} // tie-break: longer code
  }
  return best;
}
function matchSpuFromPath(rel,name){
  const parts=(rel||'').split('/').filter(Boolean);
  const base=(name||'').replace(/\.[^.]+$/,'');
  const candidates=[];
  if(parts.length>1) candidates.push(parts[0]);        // 椤跺眰鏂囦欢澶瑰悕
  candidates.push(base);                               // 鏂囦欢鍚嶏紙鍘绘墿灞曞悕锛?  candidates.push(...parts.slice(1));                  // 涓棿璺緞
  for(const c of candidates){ const m=findSpuByText(c); if(m)return m; }
  return null;
}

function traverseEntry(entry,path,files){
  return new Promise(res=>{
    if(entry.isFile){entry.file(f=>{f.relativePath=path+f.name;files.push(f);res();});}
    else if(entry.isDirectory){
      const r=entry.createReader();
      r.readEntries(async es=>{for(const e of es)await traverseEntry(e,path+entry.name+'/',files);res();});
    } else res();
  });
}

function classifyFile(name){
  const base = name.replace(/\.[^.]+$/,'').toLowerCase();
  const isInner = /鍐呭寘瑁厊inner/.test(base);
  const isOuter = !isInner && /澶栧寘瑁厊鍖呰|outer|pack/.test(base);
  const hasInstall = /瀹夎|install/.test(base);
  const hasAcc = /閰嶄欢|闄勪欢|accessory|acc/.test(base);
  const hasDim = /灏哄|灏簗dim|size|measure/.test(base);
  const hasWh = /宸ュ巶|浠搢鎵樼洏|warehouse|pallet|factory/.test(base);
  let side = 'before';
  if(/鍓峾before|old|鏀瑰墠/.test(base)) side = 'before';
  else if(/鍚巪after|new|鏀瑰悗/.test(base)) side = 'after';
  if(isInner) return {section:'packaging', kind:'鍐呭寘瑁?, side};
  if(isOuter) return {section:'packaging', kind:'澶栧寘瑁?, side};
  if(hasInstall) return {section:'install'};
  if(hasAcc) return {section:'accessory'};
  if(hasDim) return {section:'dims'};
  if(hasWh) return {section:'warehouse'};
  return {section:null, reason:'鏂囦欢鍚嶆湭鍖归厤鍒颁换浣曟澘鍧楀叧閿瘝'};
}
